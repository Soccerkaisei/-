    ctx.save(); ctx.fillStyle = '#c8312b';
    roundRect(cx-size/2, cy-size/2, size, size, size*.12); ctx.fill();
    ctx.fillStyle = '#fff'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    if (text.length === 4){
      ctx.font = 'bold '+(size*.36)+'px '+JPFONT;
      ctx.fillText(text[0], cx+size*.21, cy-size*.2); ctx.fillText(text[1], cx+size*.21, cy+size*.2);

      ctx.fillText(text[0], cx, cy-size*.21); ctx.fillText(text[1], cx, cy+size*.22);

    ctx.fillStyle = '#2b2b2b'; ctx.fillText(text, x, y); ctx.restore();
  }

    ctx.save(); roundRect(x,y,w,h,r); ctx.clip();

    ctx.drawImage(img, x+(w-img.width*s)/2, y+(h-img.height*s)/2, img.width*s, img.height*s);
    ctx.restore();
