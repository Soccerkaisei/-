  var ROUND_LEN = 10;

  var levelSelect = document.getElementById('tkwLevelSelect');
  var body = document.getElementById('tkwBody');
  var result = document.getElementById('tkwResult');
  var emojiEl = document.getElementById('tkwEmoji');
  var optsEl = document.getElementById('tkwOpts');
  var feedbackEl = document.getElementById('tkwFeedback');
  var nextBtn = document.getElementById('tkwNext');
  var barEl = document.getElementById('tkwBar');
  var qnumEl = document.getElementById('tkwQnum');

  var msgEl = document.getElementById('tkwMsg');
  var retryBtn = document.getElementById('tkwRetry');
  var backBtn = document.getElementById('tkwBack');

  var currentLevel = 'easy';
  var pool = [], idx = 0, score = 0;

  function shuffle(arr){
    var a = arr.slice();
    for (var i = a.length - 1; i > 0; i--){
      var j = Math.floor(Math.random() * (i + 1));
      var t = a[i]; a[i] = a[j]; a[j] = t;
    }
    return a;
  }

  function startRound(level){
    currentLevel = level;
    pool = shuffle(wordbank[level]).slice(0, ROUND_LEN);
    idx = 0;
    score = 0;
    levelSelect.hidden = true;
    result.hidden = true;
    body.hidden = false;
    render();
  }

  document.querySelectorAll('.tkw-lbtn').forEach(function(btn){
    btn.addEventListener('click', function(){
      startRound(btn.getAttribute('data-level'));
    });
  });

  function render(){
    feedbackEl.hidden = true;
    feedbackEl.className = 'tkw-feedback';
    feedbackEl.textContent = '';
    nextBtn.hidden = true;

    var item = pool[idx];
    var correctWord = item[0];
    emojiEl.textContent = item[1];
    qnumEl.textContent = (idx + 1) + ' / ' + pool.length + ' もん';
    barEl.style.width = Math.round((idx / pool.length) * 100) + '%';

    var bank = wordbank[currentLevel];
    var others = shuffle(bank.filter(function(w){ return w[0] !== correctWord; })).slice(0, 3).map(function(w){ return w[0]; });
    var options = shuffle([correctWord].concat(others));

    optsEl.innerHTML = '';
    options.forEach(function(word){
      var b = document.createElement('button');
      b.className = 'tkw-opt';
      b.textContent = word;
      b.addEventListener('click', function(){ answer(b, word, correctWord); });
      optsEl.appendChild(b);
    });
  }

  function answer(btn, picked, correctWord){
    var buttons = optsEl.querySelectorAll('.tkw-opt');
    var enDict = {};
    wordbank[currentLevel].forEach(function(x){ enDict[x[0]] = x[2]; });
    buttons.forEach(function(b){
      b.disabled = true;
      var w = b.textContent;
      if (w === correctWord) b.classList.add('correct');
      if (b === btn && picked !== correctWord) b.classList.add('wrong');
      var reveal = document.createElement('span');
      reveal.className = 'tkw-reveal';
      reveal.textContent = enDict[w] || '';
      b.appendChild(reveal);
    });

    feedbackEl.hidden = false;
    if (picked === correctWord){
      score++;
      feedbackEl.classList.add('correct');
      feedbackEl.textContent = '◯ せいかい！';
    } else {
      feedbackEl.classList.add('wrong');
      feedbackEl.textContent = '✕ おしい！せいかいは「' + correctWord + '」';
    }
    nextBtn.hidden = false;
  }

  nextBtn.addEventListener('click', function(){
    idx++;
    if (idx < pool.length){
      render();
    } else {
      barEl.style.width = '100%';
      body.hidden = true;
      result.hidden = false;
      scoreEl.textContent = score + ' / ' + pool.length + ' もん せいかい！';
      var pct = score / pool.length;
      if (pct === 1){
        msgEl.textContent = 'ぜんもんせいかい！すごい！';
      } else if (pct >= 0.6){
        msgEl.textContent = 'よくできました！';
      } else {
        msgEl.textContent = 'また ちょうせんしてね！';
      }
    }
  });

  retryBtn.addEventListener('click', function(){
    startRound(currentLevel);
  });

  backBtn.addEventListener('click', function(){
    result.hidden = true;
    levelSelect.hidden = false;
  });
})();
