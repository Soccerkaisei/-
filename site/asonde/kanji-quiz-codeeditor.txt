<script>
const R = (kanji, kana) => `<ruby>${kanji}<rt>${kana}</rt></ruby>`;

const EASY = [
  {kanji:"山", reading:"やま", en:"Mountain"},
  {kanji:"川", reading:"かわ", en:"River"},
  {kanji:"木", reading:"き", en:"Tree"},
  {kanji:"日", reading:"ひ", en:"Sun / Day"},
  {kanji:"月", reading:"つき", en:"Moon"},
  {kanji:"水", reading:"みず", en:"Water"},
  {kanji:"土", reading:"つち", en:"Earth / Soil"},
  {kanji:"人", reading:"ひと", en:"Person"},
  {kanji:"子", reading:"こ", en:"Child"},
  {kanji:"女", reading:"おんな", en:"Woman"},
  {kanji:"男", reading:"おとこ", en:"Man"},
  {kanji:"大", reading:"だい（おおきい）", en:"Big"},
  {kanji:"小", reading:"しょう（ちいさい）", en:"Small"},
  {kanji:"上", reading:"うえ", en:"Up"},
  {kanji:"下", reading:"した", en:"Down"},
  {kanji:"中", reading:"なか", en:"Middle / Inside"},
  {kanji:"本", reading:"ほん", en:"Book"},
  {kanji:"花", reading:"はな", en:"Flower"},
  {kanji:"雨", reading:"あめ", en:"Rain"},
  {kanji:"石", reading:"いし", en:"Stone"},
];

const HARD = [
  {kanji:"天気", reading:"てんき", en:"Weather"},
  {kanji:"音楽", reading:"おんがく", en:"Music"},
  {kanji:"勉強", reading:"べんきょう", en:"Study"},
  {kanji:"世界", reading:"せかい", en:"World"},
  {kanji:"家族", reading:"かぞく", en:"Family"},
  {kanji:"学校", reading:"がっこう", en:"School"},
  {kanji:"太陽", reading:"たいよう", en:"Sun"},
  {kanji:"教室", reading:"きょうしつ", en:"Classroom"},
  {kanji:"運動", reading:"うんどう", en:"Exercise"},
  {kanji:"季節", reading:"きせつ", en:"Season"},
  {kanji:"気分", reading:"きぶん", en:"Mood"},
  {kanji:"希望", reading:"きぼう", en:"Hope"},
  {kanji:"挑戦", reading:"ちょうせん", en:"Challenge"},
  {kanji:"成長", reading:"せいちょう", en:"Growth"},
  {kanji:"笑顔", reading:"えがお", en:"Smile"},
  {kanji:"元気", reading:"げんき", en:"Energy / Well-being"},
];

let mode = 'easy';
let quiz = [];
let pool = [];
let qIndex = 0;
let score = 0;
let answered = false;

const heroEl = document.getElementById('heroBlock');
const ROUND_LEN = 10;
const $ = id => document.getElementById(id);

function shuffle(arr){
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--){
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function show(id){
  ['screen-start','screen-game','screen-result'].forEach(s => $(s).classList.toggle('hidden', s !== id));
  heroEl.classList.toggle('hidden', id === 'screen-game');
}

function startGame(level){
  mode = level;
  pool = level === 'hard' ? HARD : EASY;
  quiz = shuffle(pool).slice(0, Math.min(ROUND_LEN, pool.length));
  qIndex = 0;
  score = 0;
  show('screen-game');
  renderQuestion();
}

function renderQuestion(){
  answered = false;
  const q = quiz[qIndex];
  $('qKanji').textContent = q.kanji;
  $('qKanji').classList.remove('hidden');
  $('qEmoji').classList.add('hidden');
  $('qPrompt').textContent = 'これは なんて よむ？';
  $('progressLabel').textContent = (qIndex + 1) + ' / ' + quiz.length;
  $('progressFill').style.width = Math.round(qIndex / quiz.length * 100) + '%';
  $('feedback').textContent = '';
  $('feedback').className = 'feedback';
  $('nextBtn').classList.add('hidden');

  const others = shuffle(pool.filter(x => x.reading !== q.reading)).slice(0, 3);
  const options = shuffle([q].concat(others));
  const wrap = $('optsWrap');
  wrap.innerHTML = '';
  options.forEach(o => {
    const b = document.createElement('button');
    b.className = 'opt-btn';
    b.textContent = o.reading;
    b.addEventListener('click', () => answer(b, o, q));
    wrap.appendChild(b);
  });
}

function answer(btn, picked, q){
  if (answered) return;
  answered = true;
  const buttons = $('optsWrap').querySelectorAll('.opt-btn');
  buttons.forEach(b => {
    b.disabled = true;
    const item = pool.find(x => x.reading === b.textContent);
    if (item.reading === q.reading) b.classList.add('correct');
    if (b === btn && picked.reading !== q.reading) b.classList.add('wrong');
    const r = document.createElement('span');
    r.className = 'opt-reveal';
    r.textContent = item.kanji + '・' + item.en;
    b.appendChild(r);
  });
  const fb = $('feedback');
  if (picked.reading === q.reading){
    score++;
    fb.classList.add('ok');
    fb.textContent = '◯ せいかい！ ' + q.kanji + '（' + q.reading + '）= ' + q.en;
  } else {
    fb.classList.add('ng');
    fb.textContent = '✕ おしい！ せいかいは「' + q.reading + '」';
  }
  $('nextBtn').textContent = qIndex + 1 < quiz.length ? 'つぎへ →' : 'けっかを みる →';
  $('nextBtn').classList.remove('hidden');
}

function showResult(){
  $('progressFill').style.width = '100%';
  show('screen-result');
  $('scoreNum').textContent = score;
  document.querySelector('.result .score small').textContent = ' / ' + quiz.length + ' もん せいかい！';
  const pct = score / quiz.length;
  let medal, msg, sub;
  if (pct === 1){ medal = '🏆'; msg = 'ぜんもん せいかい！ かんじ マスター！'; sub = (mode === 'easy' ? 'つぎは「むずかしい」にも ちょうせんしてみよう！' : 'すごい！ ともだちにも おしえてあげてね。'); }
  else if (pct >= 0.7){ medal = '🥇'; msg = 'よくできました！'; sub = 'あと すこしで ぜんもん せいかい！ もういちど やってみよう。'; }
  else if (pct >= 0.4){ medal = '🥈'; msg = 'いいかんじ！'; sub = 'まちがえた かんじも、もういちど あそぶと おぼえられるよ。'; }
  else { medal = '🌱'; msg = 'ナイス チャレンジ！'; sub = 'かんじは すこしずつ おぼえれば だいじょうぶ。また あそんでね！'; }
  $('resultMedal').textContent = medal;
  $('resultMsg').textContent = msg;
  $('resultSub').textContent = sub;
  $('shareMsg').textContent = '';
}

document.querySelectorAll('.diff-btn').forEach(b => {
  b.addEventListener('click', () => startGame(b.getAttribute('data-level')));
});

$('nextBtn').addEventListener('click', () => {
  qIndex++;
  if (qIndex < quiz.length) renderQuestion(); else showResult();
});

$('retryBtn').addEventListener('click', () => show('screen-start'));

$('shareBtn').addEventListener('click', () => {
  const url = location.href.split('#')[0].split('?')[0];
  const text = 'かんじあてゲームで ' + score + ' / ' + quiz.length + ' もん せいかい！ きみも やってみて！';
  if (navigator.share){
    navigator.share({ title: 'かんじあてゲーム｜ともだちじゃぱん', text: text, url: url }).catch(() => {});
  } else if (navigator.clipboard){
    navigator.clipboard.writeText(text + ' ' + url).then(() => {
      $('shareMsg').textContent = 'リンクを コピーしたよ！ ともだちに おくってね。';
    }).catch(() => { $('shareMsg').textContent = url; });
  } else {
    $('shareMsg').textContent = url;
  }
});
</script>
