const feedbackPhrases = ["なるほど！","いいね！","その調子！","らしさが出てきた！","いいチョイス！","あと少し！","ラストスパート！"];

let currentQ = 0;
let scores = { hikarun:0, shota:0, hii:0 };
const heroEl = document.getElementById('heroBlock');

function startQuiz(){
  document.getElementById('screen-intro').classList.add('tjxquiz-hidden');
  document.getElementById('screen-quiz').classList.remove('tjxquiz-hidden');
  document.getElementById('progressWrap').style.display = 'block';
  heroEl.classList.add('tjxquiz-hidden');
  currentQ = 0;
  scores = { hikarun:0, shota:0, hii:0 };
  renderQuestion();
}

function renderQuestion(){
  const q = questions[currentQ];
  document.getElementById('qNumber').textContent = `Q${currentQ+1}`;
  document.getElementById('qTitle').textContent = q.title;
  document.getElementById('qFeedback').textContent = '';
  document.getElementById('progressLabel').textContent = `質問 ${currentQ+1} / ${questions.length}`;
  document.getElementById('progressFill').style.width = `${(currentQ)/questions.length*100}%`;

  const wrap = document.getElementById('qOptions');
  wrap.innerHTML = '';
  q.options.forEach(opt=>{
    const btn = document.createElement('button');
    btn.className='tjxquiz-option-btn';
    const emoji = teachers[opt.teacher].emoji;
    btn.innerHTML = `<span class="tjxquiz-opt-emoji">${emoji}</span><span class="tjxquiz-opt-text">${opt.text}</span>`;
    btn.onclick = ()=> answer(opt.teacher, btn);
    wrap.appendChild(btn);
  });
}

function answer(teacherKey, btnEl){
  const allBtns = document.querySelectorAll('#qOptions .tjxquiz-option-btn');
  allBtns.forEach(b=> b.disabled = true);
  btnEl.classList.add('tjxquiz-picked');

  scores[teacherKey]++;
  document.getElementById('qFeedback').textContent =
    feedbackPhrases[Math.min(currentQ, feedbackPhrases.length-1)];

  setTimeout(()=>{
    currentQ++;
    if(currentQ < questions.length){
      renderQuestion();
    } else {
      document.getElementById('progressFill').style.width = '100%';
      showResult();
    }
  }, 450);
}

function showResult(){
  document.getElementById('screen-quiz').classList.add('tjxquiz-hidden');
  document.getElementById('progressWrap').style.display = 'none';

  let top = 'hikarun';
  let max = -1;
  for(const key in scores){
    if(scores[key] > max){ max = scores[key]; top = key; }
  }
  const t = teachers[top];
  document.getElementById('resultAvatar').innerHTML = `<img decoding="async" src="${t.photo}" alt="${t.name}">`;
  document.getElementById('resultName').textContent = t.name;
  document.getElementById('resultQuote').textContent = `「${t.quote}」`;
  document.getElementById('resultDesc').textContent = t.desc;
  document.getElementById('ctaBtn').href = t.url;

  const total = questions.length;
  const order = ['hikarun','shota','hii'];
  const scoreWrap = document.getElementById('scoreWrap');
  scoreWrap.innerHTML = '';
  order.forEach(key=>{
    const tc = teachers[key];
    const pct = Math.round((scores[key] / total) * 100);
    const row = document.createElement('div');
    row.className = 'tjxquiz-score-row';
    row.innerHTML = `
      <img decoding="async" class="tjxquiz-score-photo" src="${tc.photo}" alt="${tc.name}">
      <span class="tjxquiz-score-name">${tc.name}</span>
      <span class="tjxquiz-score-bar-bg"><span class="tjxquiz-score-bar-fill" style="width:${pct}%;background:${tc.color}"></span></span>
      <span class="tjxquiz-score-pct">${pct}%</span>
    `;
    scoreWrap.appendChild(row);
  });

  document.getElementById('shareMsg').textContent = '';
  document.getElementById('screen-result').classList.remove('tjxquiz-hidden');
}

function restartQuiz(){
  document.getElementById('screen-result').classList.add('tjxquiz-hidden');
  document.getElementById('screen-intro').classList.remove('tjxquiz-hidden');
  heroEl.classList.remove('tjxquiz-hidden');
}

document.getElementById('shareBtn').addEventListener('click', async ()=>{
  const shareData = {
    title: '先生タイプ診断｜ともだちじゃぱん',
    text: '先生タイプ診断やってみたよ！あなたはどのタイプ？',
    url: window.location.href
  };
  const shareMsg = document.getElementById('shareMsg');
  if(navigator.share){
    try{ await navigator.share(shareData); }catch(e){}
  } else if(navigator.clipboard){
    try{
      await navigator.clipboard.writeText(shareData.url);
      shareMsg.textContent = 'URLを コピーしました！';
    }catch(e){ shareMsg.textContent = shareData.url; }
  } else {
    shareMsg.textContent = shareData.url;
  }
});
